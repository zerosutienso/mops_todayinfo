use scraper::{Html, Selector};use std::collections::HashMap;use std::time::Duration;use std::error::Error;use chrono::{Datelike, Local}; // 🎯 引入日期時間功能

#[tokio::main]async fn main() -> Result<(), Box<dyn Error>> {
    // ---------------------------------------------------------
    // 🎯 核心自動化：獲取今天系統日期並自動換算為民國年格式
    // ---------------------------------------------------------
    let now = Local::now();
    
    // 西元年減去 1911 即為民國年
    let tw_year = now.year() - 1911; 
    let query_year = tw_year.to_string();
    
    // 格式化月份與日期，確保其為雙碼字串（例如 "08"、"24"）
    let query_month = format!("{:02}", now.month());
    let query_day = format!("{:02}", now.day());

    println!("=== 🇹🇼 臺灣公開資訊觀測站 當日重訊自動巡檢 ===");
    println!("📅 自動鎖定今日日期（民國格式）：{} 年 {} 月 {} 日", query_year, query_month, query_day);
    // ---------------------------------------------------------

    // 1. 初始化自動記憶 Cookie 憑證的 HTTP 客戶端
    let client = reqwest::Client::builder()
        .cookie_store(true)
        .user_agent("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36")
        .timeout(Duration::from_secs(15))
        .build()
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;

    let init_url = "https://twse.com.tw";
    let api_url = "https://twse.com.tw";

    println!("\n第一階段 ➔ 🌐 正在獲取安全 Session 憑證簽章...");
    let init_resp = client.get(init_url)
        .send()
        .await
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;
        
    if !init_resp.status().is_success() {
        eprintln!("❌ 無法連線至公開資訊觀測站，狀態碼: {}", init_resp.status());
        return Ok(());
    }
    tokio::time::sleep(Duration::from_millis(600)).await;

    println!("第二階段 ➔ 🚀 攜帶自動日期參數，發送核心 POST 請求...");
    
    // 2. 100% 複製您的 HTML 核心校驗參數
    let mut params = HashMap::new();
    params.insert("step", "1");               
    params.insert("step00", "0");             
    params.insert("firstin", "ture");         // 完美復刻網頁特有錯字 "ture"
    params.insert("off", "1");                
    params.insert("TYPEK", "all");            
    
    // 注入自動生成的今日年月日
    params.insert("year", &query_year);        
    params.insert("month", &query_month);      
    params.insert("day", &query_day);          

    // 發送核心 POST 大數據請求
    let res_obj = client
        .post(api_url)
        .form(&params)
        .send()
        .await
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;

    let response_text = res_obj
        .text()
        .await
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;

    // 3. 建立本地 CSV 檔案寫入器 (帶有 UTF-8 BOM 確保 Excel 打開不亂碼)
    let csv_path = format!("mops_report_today.csv");
    let file = std::fs::File::create(&csv_path)?;
    
    use std::io::Write as _;
    let mut buffered_file = std::io::BufWriter::new(file);
    buffered_file.write_all(b"\xEF\xBB\xBF")?; 
    let mut csv_writer = csv::Writer::from_writer(buffered_file);
    csv_writer.write_record(&["發言日期", "發言時間", "公司代號", "公司簡稱", "主旨"])?;

    println!("\n=================================== 📊 成功獲取今日重大訊息 ===================================");
    println!("{:<10}\t{:<10}\t{:<8}\t{:<10}\t{}", "發言日期", "發言時間", "公司代號", "公司簡稱", "主旨");
    println!("---------------------------------------------------------------------------------------------");

    // 4. 載入 DOM 樹進行網頁數據清洗
    let document = Html::parse_document(&response_text);
    let row_selector = Selector::parse("table tr").unwrap();
    let cell_selector = Selector::parse("td").unwrap();
    let mut found_count = 0;

    for row in document.select(&row_selector) {
        let cells: Vec<String> = row
            .select(&cell_selector)
            .map(|cell| {
                cell.text()
                    .collect::<Vec<_>>()
                    .join("")
                    .replace("&nbsp;", "")
                    .trim()
                    .to_string()
            })
            .collect();

        // 確保基本數據欄位長度符合重訊列特徵
        if cells.len() >= 5 {
            let date = &cells[0];     
            let time = &cells[1];     
            let code = &cells[2];     
            let name = &cells[3];     
            let subject = &cells[4];  

            // 🎯 核心驗證：確保第一欄有日期斜線，且第三欄公司代號為純數字（過濾掉表格標頭欄位）
            if date.contains('/') && code.chars().all(|c| c.is_digit(10)) && !code.is_empty() {
                
                // 同步寫入 CSV
                csv_writer.write_record(&[date, time, code, name, subject])?;

                // 限制終端機畫面上的主旨顯示長度，保持排版整齊
                let truncated_subject = if subject.chars().count() > 35 {
                    format!("{}...", subject.chars().take(35).collect::<String>())
                } else {
                    subject.to_string()
                };

                // 完美在畫面上輸出
                println!("{}\t{}\t{}\t{:<10}\t{}", date, time, code, name, truncated_subject);
                found_count += 1;
            }
        }
    }

    csv_writer.flush()?;

    println!("---------------------------------------------------------------------------------------------");
    println!("📈 今日數據提取完畢。全市場共成功篩選並自動導出了 {} 筆即時重大訊息紀錄。", found_count);
    if found_count > 0 {
        println!("💾 今日試算表報表已自動儲存至本地：{}", csv_path);
    } else {
        println!("💡 提示：今日回傳 0 筆。若今天為週末非交易日屬正常現象。");
    }
    println!("=============================================================================================\n");

    Ok(())
}

為了實現每 10 分鐘自動重複執行，並且只抓取最新更新項目（不重複列印與寫入舊資料），我們可以使用：

   1. tokio::time::sleep(Duration::from_secs(600)) 來建立 10 分鐘的定時循環。
   2. 在記憶體中使用一個 HashSet 集合，記錄已經抓取過重訊的唯一識別碼（由公司代號 + 發言時間組合而成）。
   3. 每次循環時，比對 HashSet，只針對沒看過的新項目進行終端機列印，並以「附加模式（Append）」即時寫入 CSV 檔案。

請維持您現有的依賴套件設定即可（需要 tokio, reqwest, scraper, csv, chrono）。
## 💻 2. 定時增量監控完全體 src/main.rs

#[tokio::main]async fn main() -> Result<(), Box<dyn Error>> {
    // 🎯 初始化一個雜湊集合，用來記憶已經抓取過的重訊唯一識別碼 (Key)
    let mut visited_records: HashSet<String> = HashSet::new();
    let csv_path = "mops_monitor_report.csv";

    // 初始化 CSV 檔案與表頭（如果檔案不存在就建立並寫入表頭；已存在就跳過）
    if std::path::Path::new(csv_path).exists() == false {
        let file = std::fs::File::create(csv_path)?;
        let mut buffered_file = std::io::BufWriter::new(file);
        use std::io::Write as _;
        buffered_file.write_all(b"\xEF\xBB\xBF")?; // 寫入 Excel 不亂碼的 BOM 頭
        let mut csv_writer = csv::Writer::from_writer(buffered_file);
        csv_writer.write_record(&["發言日期", "發言時間", "公司代號", "公司簡稱", "主旨"])?;
        csv_writer.flush()?;
    }

    println!("=== 🇹🇼 臺灣公開資訊觀測站 10分鐘定時增量監控系統啟動 ===");
    println!("💾 監控數據將實時追加儲存至本地：{}\n", csv_path);

    // 🎯 核心無限循環：每 10 分鐘執行一次
    loop {
        // 1. 動態獲取當前最新的系統日期（自動換算民國年）
        let now = Local::now();
        let query_year = (now.year() - 1911).to_string();
        let query_month = format!("{:02}", now.month());
        let query_day = format!("{:02}", now.day());

        println!("🕒 [{}] 正在發起常規巡檢 (鎖定日期: {}/{}/{})...", now.format("%Y-%m-%d %H:%M:%S"), query_year, query_month, query_day);

        // 2. 執行爬取任務
        match fetch_and_parse_mops(&query_year, &query_month, &query_day, &mut visited_records, csv_path).await {
            Ok(new_count) => {
                if new_count > 0 {
                    println!("✨ 巡檢完畢！本次發現並更新了 {} 筆全新的重大訊息！", new_count);
                } else {
                    println!("😴 巡檢完畢。全市場暫無新重大訊息更新。");
                }
            },
            Err(e) => eprintln!("⚠️ 本次巡檢發生網路或解析錯誤: {}，系統將於下個週期重試。", e),
        }

        // 3. 🎯 定時器核心：執行緒休眠 10 分鐘 (600 秒)
        println!("💤 進入休眠，10 分鐘後將自動進行下一次追蹤...\n-----------------------------------------------------");
        tokio::time::sleep(Duration::from_secs(600)).await;
    }
}

    month: &str, 
    day: &str,

    csv_path: &str

    // 🎯 核心修正：以「追加模式 (Append)」打開本地 CSV 檔案
    let file = OpenOptions::new().write(true).append(true).open(csv_path)?;
    let mut csv_writer = csv::Writer::from_writer(file);

    let mut new_records_count = 0;

    for row in document.select(&row_selector) {

            .collect();

        if cells.len() >= 5 {

            let time = &cells[1];     
            let code = &cells[2];     
            let name = &cells[3];     
            let subject = &cells[4];  

            if date.contains('/') && code.chars().all(|c| c.is_digit(10)) && !code.is_empty() {
                // 🎯 建立這筆重訊的「唯一識別碼」(由 公司代號 + 發言時間 組合)
                // 這樣可以精準防止同一家公司在不同時間發重訊被漏掉，或同時間不同公司被過濾
                let unique_key = format!("{}_{}", code, time);

                // 🎯 增量過濾核心：如果這個識別碼之前「沒抓過」才處理
                if visited.contains(&unique_key) == false {
                    // 將識別碼存入已訪問集合
                    visited.insert(unique_key);

                    // 寫入 CSV
                    csv_writer.write_record(&[date, time, code, name, subject])?;

                    // 畫面上精簡列印
                    let truncated_subject = if subject.chars().count() > 35 {
                        format!("{}...", subject.chars().take(35).collect::<String>())
                    } else {
                        subject.to_string()
                    };
                    println!(" 🆕 [新重訊] {}\t{}\t{}\t{:<10}\t{}", date, time, code, name, truncated_subject);
                    
                    new_records_count += 1;
                }
            }
        }
    }

    csv_writer.flush()?;
    Ok(new_records_count)
}

/// 核心爬取、清洗與增量過濾函數async fn fetch_and_parse_mops(
    year: &str, 
    month: &str, 
    day: &str, 
    visited: &mut HashSet<String>,
    csv_path: &str
) -> Result<usize, Box<dyn Error>> {
    // 初始化自動記憶 Cookie 憑證的核心 HTTP 客戶端，並帶有瀏覽器防禦標頭
    let client = reqwest::Client::builder()
        .cookie_store(true)
        .user_agent("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36")
        .timeout(Duration::from_secs(15))
        .build()
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;

    // 🎯 修正核心 3：使用 100% 正確的台灣官方正式主域，避開 net::ERR_NAME_NOT_RESOLVED 導航缺陷
    let init_url = "https://twse.com.tw";
    let api_url = "https://twse.com.tw";

    // 第一階段：獲取 Session 憑證
    let init_resp = client.get(init_url).send().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    if !init_resp.status().is_success() {
        return Err(format!("無法連線至網頁端，狀態碼: {}", init_resp.status()).into());
    }
    tokio::time::sleep(Duration::from_millis(600)).await;

    // 第二階段：發送完備的控制項 POST 表單參數
    let mut params = HashMap::new();
    params.insert("step", "1");               
    params.insert("step00", "0");             
    params.insert("firstin", "ture"); // 完美對齊網頁原始碼特有的拼字錯誤 "ture"
    params.insert("off", "1");                
    params.insert("TYPEK", "all");            
    
    // 傳遞經由時區換算出的精準參數，且值設定為空字串，防止後端伺服器噴出「年度未輸入」錯誤
    params.insert("year", "");        
    params.insert("month", "");      
    params.insert("day", "");          

    let res_obj = client.post(api_url).form(&params).send().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    let response_text = res_obj.text().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    
    // 載入 DOM 樹進行精準的網頁數據格子清洗
    let document = Html::parse_document(&response_text);
    let row_selector = Selector::parse("tr").unwrap();
    let cell_selector = Selector::parse("td").unwrap();

    // 🎯 以「追加模式 (Append)」打開本地 CSV 檔案，確保新資料能順暢黏貼在最尾端
    let file = OpenOptions::new().write(true).append(true).open(csv_path)?;
    let mut csv_writer = csv::Writer::from_writer(file);

    let mut new_records_count = 0;

    for row in document.select(&row_selector) {
        let cells: Vec<String> = row
            .select(&cell_selector)
            .map(|cell| {
                // 自動穿透所有 <pre> 嵌套標籤，並將隱形轉義字元 &nbsp; 徹底拔除
                cell.text()
                    .collect::<Vec<_>>()
                    .join("")
                    .replace("&nbsp;", "")
                    .trim()
                    .to_string()
            })
            .collect();

        // 完美對齊您貼出的表格特徵格式
        if cells.len() >= 5 {
            let date = &cells[0];     // 發言日期 (例: 115/08/24)
            let time = &cells[1];     // 發言時間 (例: 16:04:51)
            let code = &cells[2];     // 公司代號 (例: 7850)
            let name = &cells[3];     // 公司簡稱 (例: 寶泰生醫)
            let subject = &cells[4];  // 主旨

            // 嚴格數據列特徵驗證：確保第一欄含有日期斜線，第三欄為純數字代號
            if date.contains('/') && code.chars().all(|c| c.is_digit(10)) && !code.is_empty() {
                
                // 🎯 建立這筆重訊在全台市場上的「唯一識別碼」
                let unique_key = format!("{}_{}", code, time);

                // 🎯 增量過濾核心：不論是歷史讀出來的、還是當下巡檢看過的，只要 HashSet 沒有才處理
                if !visited.contains(&unique_key) {
                    // 將新的識別碼立刻存入已訪問集合
                    visited.insert(unique_key);

                    // 實時將此全新項目寫入 CSV 檔案
                    csv_writer.write_record(&[date, time, code, name, subject])?;

                    // 限制終端機螢幕上的主旨顯示長度，防止排版錯位
                    let truncated_subject = if subject.chars().count() > 35 {
                        format!("{}...", subject.chars().take(35).collect::<String>())
                    } else {
                        subject.to_string()
                    };
                    
                    // 在螢幕上亮出最新出爐的重大訊息項目
                    println!(" 🆕 [新重訊發現] {}\t{}\t{}\t{:<10}\t{}", date, time, code, name, truncated_subject);
                    
                    new_records_count += 1;
                }
            }
        }
    }

    // 將快取強制刷入硬碟，確保斷電資料不遺失
    csv_writer.flush()?;
    Ok(new_records_count)
}

async fn fetch_and_parse_mops(
    visited: &mut HashSet<String>,
    csv_path: &str,
    subscribed_stocks: &HashSet<&str>
) -> Result<usize, Box<dyn Error>> {
    let client = reqwest::Client::builder()
        .cookie_store(true)
        .user_agent("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36")
        .timeout(Duration::from_secs(15))
        .build()
        .map_err(|e| Box::new(e) as Box<dyn Error>)?;

    let init_url = "https://twse.com.tw";
    let api_url = "https://twse.com.tw";

    // 第一階段：獲取 Session Cookie
    let init_resp = client.get(init_url).send().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    if !init_resp.status().is_success() {
        return Err(format!("無法連線至網頁端，狀態碼: {}", init_resp.status()).into());
    }
    tokio::time::sleep(Duration::from_millis(600)).await;

    // 第二階段：發送 POST 請求要資料
    let mut params = HashMap::new();
    params.insert("step", "1");               
    params.insert("step00", "0");             
    params.insert("firstin", "ture"); // 復刻網頁原始碼特有錯字
    params.insert("off", "1");                
    params.insert("TYPEK", "all");            
    params.insert("year", "");        
    params.insert("month", "");      
    params.insert("day", "");          

    let res_obj = client.post(api_url).form(&params).send().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    let response_text = res_obj.text().await.map_err(|e| Box::new(e) as Box<dyn Error>)?;
    
    let document = Html::parse_document(&response_text);
    let row_selector = Selector::parse("tr").unwrap();
    let cell_selector = Selector::parse("td").unwrap();

    let file = OpenOptions::new().write(true).append(true).open(csv_path)?;
    let mut csv_writer = csv::Writer::from_writer(file);

    let mut new_records_count = 0;

    for row in document.select(&row_selector) {
        let cells: Vec<String> = row
            .select(&cell_selector)
            .map(|cell| {
                cell.text().collect::<Vec<_>>().join("").replace("&nbsp;", "").trim().to_string()
            })
            .collect();

        if cells.len() >= 5 {
            let date = &cells[0];     
            let time = &cells[1];     
            let code = &cells[2];     
            let name = &cells[3];     
            let subject = &cells[4];  

            if date.contains('/') && code.chars().all(|c| c.is_digit(10)) && !code.is_empty() {
                let unique_key = format!("{}_{}", code, time);

                if !visited.contains(&unique_key) {
                    visited.insert(unique_key.clone());

                    // 實時追加寫入 CSV (檔案裡保持正常純文字)
                    csv_writer.write_record(&[date, time, code, name, subject])?;

                    let truncated_subject = if subject.chars().count() > 35 {
                        format!("{}...", subject.chars().take(35).collect::<String>())
                    } else {
                        subject.to_string()
                    };
                    
                    // 🎯 核心新增：檢查此股票代號是否有被訂閱？
                    if subscribed_stocks.contains(code.as_str()) {
                        // 🌟 ANSI 高亮效果：\x1b[30;43m 代表黑字黃底，\x1b[0m 代表恢復預設
                        println!(
                            "\x1b[30;43m 🚨 [訂閱個股新重訊!!] {}\t{}\t{}\t{:<10}\t{} \x1b[0m",
                            date, time, code, name, truncated_subject
                        );
                    } else {
                        // 常規普通顯示
                        println!(" 🆕 [新重訊] {}\t{}\t{}\t{:<10}\t{}", date, time, code, name, truncated_subject);
                    }
                    
                    new_records_count += 1;
                }
            }
        }
    }

    csv_writer.flush()?;
    Ok(new_records_count)
}

use scraper::{Html, Selector};use std::collections::{HashMap, HashSet};use std::time::Duration;use std::error::Error;use std::fs::{self, OpenOptions};use std::path::Path;use chrono::{Datelike, Local, Timelike};

#[tokio::main]async fn main() -> Result<(), Box<dyn Error>> {

        let check_date_str = format!("{}_{}_{}", query_year, query_month, query_day);

        // 跨日自動更換檔案與清空快取
        if current_date_str != check_date_str {
            println!("\n📅 ──────────────────────────────────────────────────");
            if !current_date_str.is_empty() {
                println!("🌅 偵測到跨日！自動清空昨日記憶...");
                visited_records.clear(); 
            }
            current_date_str = check_date_str;
            println!("📅 今日鎖定日期：民國 {} 年 {} 月 {} 日", query_year, query_month, query_day);
            println!("────────────────────────────────────────────────────\n");
        }

        // 檔名自動依需求綁定當天年月日
        let csv_path = format!("{}_{}_{}_{}.csv", csv_path_prefix, query_year, query_month, query_day);

        // 冷啟動記憶功能
        if visited_records.is_empty() && Path::new(&csv_path).exists() {

            let mut csv_reader = csv::Reader::from_reader(file);
            let mut history_count = 0;

                        let code = record.get(2).unwrap_or("").trim();

                        visited_records.insert(unique_key);
                        history_count += 1;
                    }
                }
            }
            println!("✅ 索引重建成功！已同步過濾 {} 筆今日已抓取訊息。", history_count);
        }

        // 初始化當天專屬的 CSV 檔案 (加入防亂碼頭)
        if !Path::new(&csv_path).exists() {
            let file = std::fs::File::create(&csv_path)?;
            let mut buffered_file = std::io::BufWriter::new(file);
            use std::io::Write as _;
            buffered_file.write_all(b"\xEF\xBB\xBF")?;

/// 輔助函數：讀取設定檔並過濾掉空白行與註解fn load_config_file(file_name: &str) -> HashSet<String> {
    let mut set = HashSet::new();
    if let Ok(content) = fs::read_to_string(file_name) {
        for line in content.lines() {
            let trimmed = line.trim();
            // 過濾掉空白行或以 # 開頭的註解行
            if !trimmed.is_empty() && !trimmed.starts_with('#') {
                set.insert(trimmed.to_string());
            }
        }
    } else {
        // 如果檔案不存在，自動建立一個空的預設檔案，方便使用者未來修改
        let _ = fs::write(file_name, "# 請在此處每行輸入一個項目\n");
        println!("⚠️ 找不到 {}，已自動為您建立初始設定檔範本。", file_name);
    }
    set
}
